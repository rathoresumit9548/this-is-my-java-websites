!DOCTYPE Abstract

abstract class Animal{
abstract void sound();
}
class dog extends Animal{
void sound(){
System.out.println("dog barks"); }
}
class cat extends Animal{
void sound(){
System.out.println("cat maun"); }
}
public class Main{
public static void main(String[] args){
dog d1 = new dog();
d1.sound();
cat c1 = new cat();
c1.sound(); }
}
