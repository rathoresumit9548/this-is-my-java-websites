abstract class Animal{
abstract void sound();
}
class dog extends Animal{
void sound(){
System.out.println("dog barks");
}
}
class cat extends Animal{
void sound(){
System.out.println("cat maun");
}
}
public class Main{
public static void main(String[] args){
Animal p1;
p1 = new dog();
p1.sound();
p1 = new cat();
p1.sound();
}
}
